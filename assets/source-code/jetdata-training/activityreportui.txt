<script>
    const API_URL = "https://demo.jetdata.ai/ims/jetapi.php";
    const API_KEY = "<INSERT JETDATA API KEY HERE>";
    const PROJECT = "INSER PROJECT NAME HERE";
    const FORM_ID = "INSERT FORM ID HERE";

    const fileIn = document.getElementById('fileIn');
    const imgPrev = document.getElementById('imgPrev');
    const dropText = document.getElementById('dropText');
    const upBtn = document.getElementById('upBtn');

    fileIn.addEventListener('change', function() {
        const file = this.files[0];
        if (file) {
            dropText.style.display = "none";
            imgPrev.style.display = "block";
            if (file.type.startsWith('image/')) {
                const reader = new FileReader();
                reader.onload = e => imgPrev.src = e.target.result;
                reader.readAsDataURL(file);
            } else {
                imgPrev.src = "https://cdn-icons-png.flaticon.com/512/337/337946.png";
            }
        }
    });

    async function uploadFile() {
        if (!fileIn.files[0]) return alert("Please select a file.");
        upBtn.disabled = true;
        const fd = new FormData();
        fd.append('activityreport_image', fileIn.files[0]);

        try {
            const resp = await fetch(`${API_URL}?method=addRecord&project=${PROJECT}&id_form=${FORM_ID}&key=${API_KEY}`, {
                method: 'POST', body: fd
            });
            const res = await resp.json();
            if (res.status === 'success') {
                showMsg("✅ Upload Success!", "success");
                getRecords();
            } else { throw new Error(res.error); }
        } catch (e) { showMsg("❌ Error: " + e.message, "error"); }
        finally { upBtn.disabled = false; }
    }

    async function getRecords() {
        const viewBtn = document.getElementById('viewBtn');
        viewBtn.innerText = "Loading...";
        try {
            const resp = await fetch(`${API_URL}?method=getRecords&project=${PROJECT}&id_form=${FORM_ID}&key=${API_KEY}`);
            const res = await resp.json();
            let data = res.data || res;
            const records = Array.isArray(data) ? data : Object.values(data).filter(i => typeof i === 'object');

            if (records.length > 0) {
                renderTable(records);
                document.getElementById('recordsCard').style.display = "block";
                document.getElementById('tableWrapper').style.display = "block";
            }
        } catch (e) { showMsg("❌ Connection Error", "error"); }
        finally { viewBtn.innerText = "View All Records"; }
    }

    function renderTable(data) {
        const tbody = document.getElementById('tableBody');
        tbody.innerHTML = '';
        [...data].reverse().forEach((row, index) => {
            tbody.innerHTML += `
                <tr>
                    <td style="color:#94a3b8"><b>${index + 1}</b></td>
                    <td><b>${row.customer_name || '-'}</b></td>
                    <td class="wrap-cell">${row.customer_address || '-'}</td>
                    <td>${row.activity_type || '-'}</td>
                    <td>${row.product_name || '-'}</td>
                    <td>${row.serial_no || '-'}</td>
                    <td>${row.engineer_name || '-'}</td>
                    <td>${row.start_datetime || '-'}</td>
                    <td>${row.finish_datetime || '-'}</td>
                    <td class="wrap-cell">${row.symptom_ar || '-'}</td>
                    <td class="wrap-cell">${row.action_ar || '-'}</td>
                    <td><span style="font-weight:600; color:${row.approval === 'Approved' ? '#166534' : '#9a3412'}">${row.approval || 'Pending'}</span></td>
                </tr>
            `;
        });
    }

    function filterTable() {
        const q = document.getElementById('searchIn').value.toLowerCase();
        const rows = document.querySelectorAll('#tableBody tr');
        rows.forEach(r => r.style.display = r.innerText.toLowerCase().includes(q) ? '' : 'none');
    }

    function showMsg(m, t) {
        const s = document.getElementById('status');
        s.innerText = m; s.className = t; s.style.display = "block";
    }
</script>
