<script>
const GET_RECORDS_URL = "INSERT JETDATA GET RECORD API HERE";
const SEARCH_RECORD_URL = "INSERT JETDATA SEARCH RECORD HERE";

async function searchRecords() {
  try {
    const res = await fetch(GET_RECORDS_URL);
    const data = await res.json();
    let records = Array.isArray(data) ? data : (data.data ? (Array.isArray(data.data) ? data.data : Object.values(data.data)) : []);
    renderTable(records);
  } catch (e) { console.error("Fetch error", e); }
}

function renderTable(records) {
  if (!records.length) { document.getElementById("tableContainer").innerHTML = "No records found"; return; }
  const headers = Object.keys(records[0]);
  let html = "<table><thead><tr>";
  headers.forEach(h => html += `<th>${h}</th>`);
  html += "</tr></thead><tbody>";
  records.forEach(r => {
    html += `<tr onclick="selectRow('${r.S_ROW_ID || r.ROW_ID || ""}')">`;
    headers.forEach(h => html += `<td>${r[h] ?? ""}</td>`);
    html += "</tr>";
  });
  html += "</tbody></table>";
  document.getElementById("tableContainer").innerHTML = html;
}

function selectRow(id) { if (id) document.getElementById("rowIdInput").value = id; }

async function generateOvertime() {
  const rowId = document.getElementById("rowIdInput").value.trim();
  if (!rowId) { alert("Please enter or select an ID"); return; }

  // We search for S_ROW_ID specifically
  const url = `${SEARCH_RECORD_URL}&search_field=S_ROW_ID&search_value=${encodeURIComponent(rowId)}`;
  
  try {
    const res = await fetch(url);
    const data = await res.json();
    
    // Normalize data structure
    let results = [];
    if (data.data) {
        results = Array.isArray(data.data) ? data.data : Object.values(data.data);
    }

    // Attempt to find the exact match, otherwise take the first result returned by the search
    const record = results.find(r => String(r.S_ROW_ID) === String(rowId) || String(r.ROW_ID) === String(rowId)) || results[0];

    if (!record) {
      document.getElementById("markdownPreview").innerHTML = `<p style="color:red">Record ${rowId} not found in database.</p>`;
      return;
    }

    const startDate = record.start_datetime ? record.start_datetime.split(' ')[0] : "";
    const startTime = record.start_datetime ? record.start_datetime.split(' ')[1] : "";
    const endTime = record.finish_datetime ? record.finish_datetime.split(' ')[1] : "";

    const markdown = `
## Form Overtime Company

**Tahun/Bulan/Hari** : ${record.created_date || ""}  
**Nama Karyawan** : ${record.engineer_name || ""}

| No. | Nama Customer | Hari/Tanggal | Mulai | Selesai | Keterangan |
|:---:|:---|:---|:---:|:---:|:---|
| 1 | ${record.customer_name || ""} | ${startDate} | ${startTime} | ${endTime} | ${record.description_ar || ""} |

<div class="signature-grid">
    <div>
        <p>Diajukan oleh,</p>
        <br><br><br>
        <p>____________________</p>
    </div>
    <div>
        <p>Disetujui Oleh,</p>
        <br><br><br>
        <p>____________________</p>
    </div>
</div>
`;
    document.getElementById("markdownPreview").innerHTML = marked.parse(markdown);
  } catch (err) {
    console.error(err);
    document.getElementById("markdownPreview").innerHTML = "Error fetching record.";
  }
}
