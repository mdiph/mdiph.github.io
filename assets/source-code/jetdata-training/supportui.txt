<script>
    /* ============================================================
       CONFIGURATION & VARIABLES
       ============================================================ */
    // Using the webhook from your uploaded file
    const WEBHOOK_URL = '<INSERT YOUR N8N WEBHOOK CHAT HERE>';
    
    // Global variables from your environment
    const SESSION_KEY = window.session_key || (typeof session_key !== 'undefined' ? session_key : `sess_${Date.now()}`);
    const PROJECT = window.project || (typeof project !== 'undefined' ? project : '');

    // Default User Object
    let currentUser = { 
        fullName: 'Guest', 
        firstName: '', 
        lastName: '', 
        email: '', 
        userId: '' 
    };

    /* ============================================================
       USER DATA FETCHING (Logic from your uploaded file)
       ============================================================ */
    async function loadUserInfo() {
        try {
            const formData = new FormData();
            formData.append('method', 'getUserInfo');
            formData.append('project', PROJECT);
            formData.append('session_key', SESSION_KEY);
            
            // Construct path dynamically as per your code
            const host = window.location.origin;
            const path = window.location.pathname.split('/')[1] || 'ims';
            const apiUrl = `${host}/${path}/api_v3/api.php`;

            console.log("Fetching user info from:", apiUrl); // Debug log

            const res = await fetch(apiUrl, { method: 'POST', body: formData });
            
            // Robust JSON parsing from your code (handles prefixes)
            const text = await res.text();
            const jsonStart = text.search(/[\[{]/);
            const result = JSON.parse(jsonStart > 0 ? text.substring(jsonStart) : text);

            if (result.status === 'success' && result.user) {
                currentUser = {
                    fullName: `${result.user.first_name || ''} ${result.user.last_name || ''}`.trim() || result.user.username || 'User',
                    firstName: result.user.first_name || '',
                    lastName: result.user.last_name || '',
                    email: result.user.email || '',
                    userId: result.user.id_user || ''
                };
            }
        } catch (e) {
            console.log('User info fetch failed (likely running locally without PHP backend). Using Guest.');
        }
        
        // Update UI
        updateUserHeader();
    }

    function updateUserHeader() {
        const nameEl = document.getElementById('userName');
        const avatarEl = document.getElementById('userAvatar');
        
        nameEl.textContent = currentUser.fullName;
        
        // Initials logic
        const parts = currentUser.fullName.split(' ').filter(p => p);
        const initials = parts.length >= 2 
            ? (parts[0][0] + parts[parts.length-1][0]).toUpperCase() 
            : parts[0][0].toUpperCase();
            
        avatarEl.textContent = initials;
    }

    /* ============================================================
       CHAT LOGIC
       ============================================================ */
    const chatBox = document.getElementById('chatBox');
    const userInput = document.getElementById('userInput');
    const sendBtn = document.getElementById('sendBtn');
    const loading = document.getElementById('loading');

    // Setup Markdown
    if (typeof marked !== 'undefined') {
        marked.setOptions({ breaks: true, gfm: true });
    }

    function addMessage(text, sender, isMarkdown = false) {
        const msgDiv = document.createElement('div');
        msgDiv.className = `message ${sender}`;
        
        const bubble = document.createElement('div');
        bubble.className = 'bubble';

        if (sender === 'bot' && isMarkdown && typeof marked !== 'undefined') {
            bubble.innerHTML = marked.parse(text);
        } else {
            // Simple text escape for user messages to prevent XSS
            bubble.textContent = text; 
        }

        msgDiv.appendChild(bubble);
        chatBox.insertBefore(msgDiv, loading); // Insert before loading dots
        scrollToBottom();
    }

    function scrollToBottom() {
        chatBox.scrollTop = chatBox.scrollHeight;
    }

    async function handleSend() {
        const text = userInput.value.trim();
        if (!text) return;

        // 1. Add User Message
        addMessage(text, 'user');
        userInput.value = '';
        userInput.style.height = 'auto'; // Reset height
        
        // 2. UI State: Loading
        loading.style.display = 'block';
        scrollToBottom();
        sendBtn.disabled = true;

        try {
            // 3. Send Payload (Using structure from your uploaded code)
            const payload = {
                chatInput: text,
                sessionId: SESSION_KEY,
                userName: currentUser.fullName,
                userFirstName: currentUser.firstName,
                userLastName: currentUser.lastName,
                userEmail: currentUser.email,
                userId: currentUser.userId,
                sessionKey: SESSION_KEY,
                project: PROJECT,
                timestamp: new Date().toISOString()
            };

            const response = await fetch(WEBHOOK_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) throw new Error(`HTTP ${response.status}`);

            // 4. Handle Response
            const data = await response.json();
            
            // Extract text logic from your uploaded code
            let botResponse = '';
            if (typeof data === 'string') botResponse = data;
            else if (Array.isArray(data) && data.length) botResponse = JSON.stringify(data[0]);
            else botResponse = data.output || data.text || data.message || data.response || JSON.stringify(data);

            // 5. Add Bot Message
            loading.style.display = 'none';
            addMessage(botResponse, 'bot', true); // Enable markdown for bot

        } catch (error) {
            console.error(error);
            loading.style.display = 'none';
            addMessage("Sorry, I'm having trouble connecting right now.", 'bot');
        }

        sendBtn.disabled = false;
        userInput.focus();
    }

    /* ============================================================
       EVENT LISTENERS & INIT
       ============================================================ */
    sendBtn.addEventListener('click', handleSend);

    userInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            handleSend();
        }
    });

    // Auto-resize textarea
    userInput.addEventListener('input', function() {
        this.style.height = 'auto';
        this.style.height = (this.scrollHeight) + 'px';
    });

    // Initialize
    loadUserInfo();

</script>
